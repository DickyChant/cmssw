cmake_minimum_required(VERSION 3.15)
project(CMSDirePlugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_path(PYTHIA_INCLUDE Pythia8/Pythia.h HINTS "${PYTHIA_ROOT}/include" REQUIRED)
find_library(PYTHIA_LIBRARY pythia8 HINTS "${PYTHIA_ROOT}/lib" REQUIRED)
# Deliberately expose ONLY old Dire headers. All Pythia core headers and ABI
# must come from the current release, never from the 8.315 source tree.
file(GLOB DIRE_HEADERS "${DIRE_SOURCE}/include/Pythia8/Dire*.h")
file(GLOB DIRE_SOURCES "${DIRE_SOURCE}/src/Dire*.cc")
if(NOT DIRE_HEADERS OR NOT DIRE_SOURCES)
  message(FATAL_ERROR "DIRE_SOURCE must point at the pinned Pythia 8.315 source")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/Pythia8")
file(COPY ${DIRE_HEADERS} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/Pythia8")
add_library(CMSDire SHARED DirePlugin.cc ${DIRE_SOURCES})
target_include_directories(CMSDire PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include" "${PYTHIA_INCLUDE}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/share/CMSDire")
foreach(xml DireShowers.xml DireExpert.xml DireWeights.xml)
  configure_file("${DIRE_SOURCE}/share/Pythia8/xmldoc/${xml}"
                 "${CMAKE_CURRENT_BINARY_DIR}/share/CMSDire/${xml}" COPYONLY)
  install(FILES "${DIRE_SOURCE}/share/Pythia8/xmldoc/${xml}" DESTINATION share/CMSDire)
endforeach()
target_link_libraries(CMSDire PRIVATE "${PYTHIA_LIBRARY}" ${CMAKE_DL_LIBS})
target_link_options(CMSDire PRIVATE -Wl,--no-undefined)
install(TARGETS CMSDire LIBRARY DESTINATION lib)
install(FILES "${DIRE_SOURCE}/COPYING" "${DIRE_SOURCE}/GUIDELINES" DESTINATION share/CMSDire)
